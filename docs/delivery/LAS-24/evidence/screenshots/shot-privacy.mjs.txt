// Usage: node shot-privacy.mjs <appUrl> <apiUrl> <outDir> — library, consent and delete-account screenshots.
import { randomBytes } from 'node:crypto';
import { chromium } from '/home/user/MPS-Audio-Studio-delivery/node_modules/@playwright/test/index.mjs';
import { makeWav } from '/home/user/MPS-Audio-Studio-delivery/tests/fixtures/wav.js';
const [app, apiUrl, out] = process.argv.slice(2);
const browser = await chromium.launch({ executablePath: '/opt/pw-browsers/chromium-1194/chrome-linux/chrome' });
const page = await browser.newPage({ viewport: { width: 1280, height: 900 } });
const email = `demo-${randomBytes(3).toString('hex')}@example.test`;
const password = randomBytes(12).toString('base64url');
const dialog = page.getByRole('dialog');
const secret = async (locator, value) => locator.evaluate((input, v) => { Object.getOwnPropertyDescriptor(HTMLInputElement.prototype, 'value').set.call(input, v); input.dispatchEvent(new Event('input', { bubbles: true })); }, value);
await page.goto(app);
await page.getByRole('button', { name: 'Account' }).click();
await dialog.getByRole('button', { name: 'Create account' }).click();
await dialog.getByLabel('Email').fill(email);
await secret(dialog.getByLabel('Password'), password);
await dialog.getByRole('button', { name: 'Create account' }).click();
await dialog.getByRole('heading', { name: 'Check your email' }).waitFor();
const mail = (await (await fetch(`${apiUrl}/api/dev/outbox?to=${encodeURIComponent(email)}`)).json()).messages[0];
await page.goto(mail.body.match(/https?:\/\/\S+/)[0]);
await dialog.getByText('Email verified. You can sign in now.').waitFor();
await dialog.getByLabel('Email').fill(email);
await secret(dialog.getByLabel('Password'), password);
await dialog.getByRole('button', { name: 'Sign in' }).click();
await dialog.getByRole('button', { name: 'Close account dialog' }).click();
await page.getByLabel('Import WAV').setInputFiles({ name: 'demo-song.wav', mimeType: 'audio/wav', buffer: makeWav({ seconds: 2 }) });
await page.getByRole('region', { name: 'Transport', exact: true }).getByRole('status').filter({ hasText: 'Ready' }).waitFor();
await page.getByRole('button', { name: 'Account' }).click();
await dialog.getByRole('button', { name: 'Save “demo-song.wav” to my library' }).click();
await dialog.getByRole('list', { name: 'Saved files' }).waitFor();
await page.screenshot({ path: `${out}/library-saved.png` });
await dialog.getByRole('button', { name: 'Send demo-song.wav to Local test recipient' }).click();
await dialog.getByRole('group').waitFor();
await page.screenshot({ path: `${out}/consent-question.png` });
await dialog.getByRole('button', { name: 'Cancel' }).click();
await dialog.getByRole('button', { name: 'Delete account…' }).click();
await dialog.getByLabel('Password to confirm').scrollIntoViewIfNeeded();
await page.screenshot({ path: `${out}/delete-account.png` });
await browser.close();
